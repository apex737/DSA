```cpp
#include <iostream>
#include <vector>

using namespace std;

int N, M;
int board[11][11];
struct Pos
{
    int r, c;
};

// 우하좌상
int dc[] = {1, 0, -1, 0};
int dr[] = {0, 1, 0, -1};

int go(Pos &st, const Pos &en, int &dir);
int main()
{
    int T;
    cin >> T;
    for (int t = 1; t <= T; t++)
    {
        cin >> N;
        vector<Pos> apples(11);
        int num = 1;
        for (int i = 0; i < N; i++)
            for (int j = 0; j < N; j++)
            {
                cin >> board[i][j];
                if (board[i][j] == num)
                {
                    // apple 숫자를 POS 좌표에 매핑
                    apples[num++] = {i, j}; //
                }
            }
        // 보드에는 1~num-1 까지 수가 들어있다.
        // 1->2, 2->3 ... 를 시뮬레이션하면서 sum 갱신
        int sum = 0;
        int dir = 0;
        Pos cur{0, 0};
        for (int i = 1; i < num; i++)
        {
            sum += go(cur, apples[i], dir);
        }

        cout << "#" << t << " " << sum << "\n";
    }
    return 0;
}

int go(Pos &st, const Pos &en, int &dir)
{
    int turns = 0;
    // 1. 위치벡터
    int vr = en.r - st.r;
    int vc = en.c - st.c;

    // 2. 진행방향 단위벡터와 내적
    int front = vr * dr[dir] + vc * dc[dir];

    // 3. 진행방향의 시계방향 단위벡터와 내적
    int rightDir = (dir + 1) % 4;
    int right = vr * dr[rightDir] + vc * dc[rightDir];

    // 4. front, right 부호를 이용해서 사분면 확정
    if(right < 0) turns = 3;             // x- 축과 2,3 사분면
    else if(front < 0) turns = 2;        // y- 축과 4사분면 (x+ 제외)
    else if(right > 0) turns = 1;        // x+ 축과 1사분면 (y+ 제외)
    else turns = 0;                      // 앞 (y+ 축)

    // 위의 코드는 간단해 보이지만, 직접 구현한다고 생각하면 경계 처리가 까다롭다.
    // 조건문의 영역을 하나씩 지우면서 범위를 좁혀나가야 함.
    
    st = en;
    dir = (dir + turns) % 4;
    return turns;
}
```
